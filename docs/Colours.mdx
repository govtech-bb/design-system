export const getContrastColor = (hexColor) => {
  const hex = hexColor.replace('#', '');
  const r = parseInt(hex.substr(0, 2), 16);
  const g = parseInt(hex.substr(2, 2), 16);
  const b = parseInt(hex.substr(4, 2), 16);
  const luminance = (0.299 * r + 0.587 * g + 0.114 * b) / 255;
  return luminance > 0.5 ? '#000000' : '#FFFFFF';
};

export const ColorSwatch = ({ color, label, hex, contrast, token }) => (
  <div className="inline-block flex-1">
    <div
      className="w-full h-20 flex items-center justify-center font-semibold text-sm"
      style={{
        backgroundColor: color,
        color: getContrastColor(color),
      }}
    >
      {contrast}
    </div>
    <div className="py-2 text-xs">
      <div className="font-semibold mb-0.5">{label}</div>
      <div style={{ color: '#6b7280', marginBottom: '2px' }}>{hex}</div>
      {token && <div style={{ color: '#6b7280', fontSize: '11px' }}>{token}</div>}
    </div>
  </div>
);

export const ColorRow = ({ colors }) => (
  <div className="flex gap-0 mb-10">
    {colors.map((color, index) => (
      <ColorSwatch key={index} {...color} />
    ))}
  </div>
);

export const SectionTitle = ({ children }) => (
  <h3 className="text-lg font-semibold mt-8 mb-4">{children}</h3>
);
